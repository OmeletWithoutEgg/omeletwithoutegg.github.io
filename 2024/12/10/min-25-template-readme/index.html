template <typename U, typename V> struct min25 {
  lld n; int sq;
  vector<U> Ss, Sl, Spre; vector<V> Rs, Rl;
  Sieve sv; vector<lld> quo;
  U &S(lld d) { return d < sq ? Ss[d] : Sl[n / d]; }
  V &R(lld d) { return d < sq ? Rs[d] : Rl[n / d]; }
  min25(lld n_) : n(n_), sq((int)sqrt(n) + 1),
    Ss(sq), Sl(sq), Spre(sq), Rs(sq), Rl(sq), sv(sq) {
    for (lld i = 1, Q; i <= n; i = n / Q + 1)
      quo.push_back(Q = n / i);
  }
  U F_prime(auto &&f, auto &&F) {
    for (lld p : sv.primes) Spre[p] = f(p);
    for (int i = 1; i < sq; i++) Spre[i] += Spre[i - 1];
    for (lld i : quo) S(i) = F(i) - F(1);
    for (lld p : sv.primes)
      for (lld i : quo) {
        if (p * p > i) break;
        S(i) -= f(p) * (S(i / p) - Spre[p - 1]);
      }
    return S(n);
  } // F_prime: \sum _ {p is prime, p <= n} f(p)
  V F_comp(auto &&g, auto &&h) {
    for (lld i : quo) R(i) = h(S(i));
    for (lld p : sv.primes | views::reverse)
      for (lld i : quo) {
        if (p * p > i) break;
        lld prod = p;
        for (int c = 1; prod * p <= i; ++c, prod *= p) {
          R(i) += g(p, c) * (R(i / prod) - h(Spre[p]));
          R(i) += g(p, c + 1);
        }
      }
    return R(n);
  } // F_comp: \sum _ {2 <= i <= n} g(i)
}; // O(n^{3/4} / log n)

Description

這個模板要解決的問題簡單來說就是有一個積性函數 ggg，計算 ∑i≤ng(i)\sum _ {i \leq n} g(i)∑i≤n​g(i)。因為是積性函數，所以只要給定質數冪的值，即 g(pc)g(p^c)g(pc)，就可以對所有正整數定義 ggg 的值。然而為了快速計算 ggg 還必須滿足一些條件，在學 min25 篩時常常會看到「質數點 ppp 必須是一個低階多項式」，不過其實可以更一般化。使用這個模板時必須要構造兩個 type U,VU, VU,V 以及函數 f,hf, hf,h 滿足以下條件：

  1. U,VU, VU,V 都是環，即在上面可以定義加法、乘法，加法可交換且有反元素（可以做減法），乘法對加法有分配律。
  2. 要求 f:N→Uf: \mathbb{N} \to Uf:N→U 是完全積性，即 ∀a,b∈N,f(ab)=f(a)f(b)\forall a, b \in\mathbb{N}, f(ab) = f(a)f(b)∀a,b∈N,f(ab)=f(a)f(b)。
  3. fff 的前綴和 F(i)=∑j≤if(i)F(i) = \sum _ {j \leq i} f(i)F(i)=∑j≤i​f(i) 可以快速求值。
  4. 對於質數 ppp，g(p)=h(f(p))g(p) = h(f(p))g(p)=h(f(p))。
  5. h:U→Vh: U \to Vh:U→V 是一個加法上的同態，即 h(u1+u2)=h(u1)+h(u2)h(u_1 + u_2) = h(u_1) + h(u_2)h(u1​+u2​)=h(u1​)+h(u2​)。
  6. 給定 p,cp, cp,c 之後，g(pc)g(p^c)g(pc) 可以快速求值。

呼叫 F_prime 會對所有 x∈Dn=⌊ni⌋∣i=1,2,…,nx \in D_n = \\{\lfloor \frac{n}{i} \rfloor | i=1,2,\dots,n \\}x∈Dn​=⌊in​⌋∣i=1,2,…,n 計算 S(x)=∑p is prime,p≤xf(p)S(x) = \sum _ {p \text{ is prime}, p \leq x} f(p)S(x)=∑p is prime,p≤x​f(p)。請注意 F(x)=∑i≤xf(i)F(x) = \sum _ {i \leq x} f(i)F(x)=∑i≤x​f(i)，和 SSS 不同。
呼叫 F_comp 會對所有 x∈Dnx \in D_nx∈Dn​ 計算 R(x)=∑2≤i≤xg(i)R(x) = \sum _ {2 \leq i \leq x} g(i)R(x)=∑2≤i≤x​g(i)。函數回傳的值就是 R(n)R(n)R(n)，也就是最一開始求的值。注意這個值不會把 g(1)g(1)g(1) 算進去。

  V F_comp_dfs(auto &&g, auto &&h) {

Caveats

這份程式碼並沒有直接要求 U,VU, VU,V 的加法單位元和乘法單位元（0,10, 10,1），但都預設空的 vector 會是 000，並且因為 f,gf, gf,g 是積性的，定義上應該要滿足 f(1)=1U,g(1)=1Vf(1) = 1_U, g(1) = 1_Vf(1)=1U​,g(1)=1V​。

另外，完全積性函數這個條件，我懷疑其實也是可以拿掉的。在第一階段（F_prime）我們會把包含小質數的數字全部扣掉，這相當於是跟 f−1f^{-1}f−1 做捲積，而當 fff 是完全積性函數，inverse 會非常好算：fp−1(x)=1−f(p)xf_p^{-1}(x) = 1 - f(p)xfp−1​(x)=1−f(p)x（見貝爾級數），若 fff 不是完全積性的，inverse 要花點力氣算。不過，我們仍然會需要構造一個前綴和很好算的積性函數 fff，這是這個演算法的大前提。把完全積性的條件改掉之後可能常數會差一點，而且當初寫的時候沒有想這麼多。

這份程式碼如果 f,g,h,Ff,g,h,Ff,g,h,F 都是 O(1)\mathcal{O}(1)O(1) 的話，大概可以跑 101010^{10}1010 到 101110^{11}1011。因為據說有 O~(N)\tilde{\mathcal{O}}(\sqrt{N})O~(N​) 的演算法了所以就去翻了一堆文章，還是連同樣條件下的 O(n23log⁡−1n)\mathcal{O}(n^{\frac{2}{3}} \log^{-1}n)O(n32​log−1n) 的版本也寫不出來。去翻 library checker 的各種 top coder 好像也都是 O~(n23)\tilde{\mathcal{O}}(n^{\frac{2}{3}})O~(n32​) 的複雜度的。只有再次學會了杜教篩，寫了一個帶 template 的模板過了 sum_of_totient_function，但跟本文的模板比也只是 192ms 跟 397ms 的差距而已。
